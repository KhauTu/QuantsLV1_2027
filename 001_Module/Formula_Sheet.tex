\subsection*{Total Return}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:total_return}
\vspace{-0.4em}
\[
\renewcommand{\arraystretch}{2.0}
\begin{array}{r c c c c}
\text{Total Return} & = & \text{Capital Appreciation Return} & + & \text{Capital Distribution Return} \\
r & = & r_{\text{price}} & + & r_{\text{distribution}} \\
  & = & \dfrac{P_1-P_0}{P_0} & + & \dfrac{\text{Inc}}{P_0}
\end{array}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Arithmetic and Geometric Holding Period Return}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:arithmetic_mean}
\vspace{-0.4em}
\[
\bar{r}_A = \frac{r_1 + r_2 + \dots + r_T}{T} = \frac{1}{T}\sum_{t=1}^{T}r_t
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:geometric_mean}
\vspace{-0.4em}
\[
\bar{r}_G = \sqrt[T]{(1 + r_1) \times (1 + r_2) \times \dots \times (1 + r_T)} - 1 = \left[ \prod_{t=1}^{T}(1+r_t) \right]^{1/T} - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Compounding: Daily, Monthly, Quarterly, and Annual Returns}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:annualized_return}
\vspace{-0.4em}
\[
r_{\text{annualized}} = (1 + r_{\text{period}})^c - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Continuous Compounding}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:continuous_compounding}
\vspace{-0.4em}
\[
\tilde{r}_{t, t+1} = \ln \left( \frac{P_{t+1}}{P_t} \right) = \ln(1 + r_{t, t+1})
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{The Required Rate of Return}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:real_risk_free}
\vspace{-0.4em}
\[
r_{\text{real\_rf}} = \frac{1 + r_{\text{rf}}}{1 + \text{infl}} - 1 \quad \text{or quick approximation: } r_{\text{real\_rf}} \approx r_{\text{rf}} - \text{infl}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:required_return}
\vspace{-0.4em}
\[
r = r_{\text{rf}} + \text{infl} + \sum_{j=1}^{n} (f_j \times \text{rp}_j)
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item $n$ is the number of risk factors
  \item $f_{j}$ is the exposure of the security to the $j^{\text{th}}$ risk factor
  \item $\text{rp}_j$ is the specific risk premium associated with the $j^{\text{th}}$ risk factor
  \item $\sum_{j=1}^{n} (f_j \times \text{rp}_j)$ is the sum of all risk premia.
\end{itemize}
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Excess Returns}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:excess_return}
\vspace{-0.4em}
\[
\dot{r}_i = \frac{1+r_i}{1+r_B} - 1 \quad \text{or quick approximation: } \dot{r}_i \approx r_i - r_B
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Real Returns}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:real_return}
\vspace{-0.4em}
\[
r_{\text{real}} = \frac{1+r_i}{1+\text{infl}} - 1 \quad \text{or quick approximation: } r_{\text{real}} \approx r_i - \text{infl}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Pre-Tax and After-Tax Nominal Return}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:after_tax_return}
\vspace{-0.4em}
\[
r_{\text{net-tax}} = r_{\text{price}} \times (1 - \text{tax}_{\text{capital gains}}) + r_{\text{capital}} \times (1 - \text{tax}_{\text{capital distribution}})
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Leveraged Returns}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:leveraged_return}
\vspace{-0.4em}
\[
R_L = R_A + \frac{D}{E}(R_A - R_D)
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item $R_L$ is the leveraged return (Return on Equity)
  \item $R_A$ is the return on total assets (unleveraged return)
  \item $R_D$ is the cost of debt (borrowing interest rate)
  \item $D/E$ is the debt-to-equity ratio.
\end{itemize}
\end{tcolorbox}

\subsection*{Time-Weighted Rate of Return (TWR)}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:twr_m3}
\vspace{-0.4em}
\[
r_{\text{TW}} = \prod_{t=1}^{T} (1 + r_t) - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:twr_G}
\vspace{-0.4em}
\[
r_{\text{G}} = \sqrt[T]{\prod_{t=1}^{T} (1 + r_t)} - 1
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item $r_{\text{TW}}$ is the time-weighted rate of return.
  \item $T$ is the total number of subperiods.
  \item $r_t$ is the holding period return for subperiod $t$, calculated between two successive cash flow events.
\end{itemize}
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Money-Weighted Rate of Return (MWR)}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:mwr_m3}
\vspace{-0.4em}
\[
0 = \sum_{t=0}^{T} \frac{\text{CF}_t}{(1 + r_{\text{MW}})^t}
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item $r_{\text{MW}}$ is the money-weighted rate of return.
  \item $\text{CF}_t$ is the net cash flow at time $t$.
\end{itemize}
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Price Return vs. Total Return Indexes}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:price_return_rate}
\vspace{-0.4em}
\[
r_{\text{PR}} = \frac{P_{t+1}}{P_t} - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:total_return_rate}
\vspace{-0.4em}
\[
r_{\text{TR}} = \frac{P_{t+1} + \text{Div}_{t+1}}{P_t} - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Index Weighting Methodologies}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:general_index_value}
\vspace{-0.4em}
\[
\text{Index Value}_t = \sum_{i=1}^{N} w_{i,t} \times P_{i,t}
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item $w_{i,t}$ is the weight of constituent $i$ at time $t$.
  \item $P_{i,t}$ is the price of constituent $i$ at time $t$.
\end{itemize}
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:market_cap_weight}
\vspace{-0.4em}
\[
w_i^M = \frac{Q_i P_i}{\sum_{j=1}^{N} Q_j P_j}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:divisor_index_value}
\vspace{-0.4em}
\[
V_t = \frac{\sum_{i=1}^{N} Q_{i,t} P_{i,t}}{D_t}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:float_adjusted_weight}
\vspace{-0.4em}
\[
w_i^{M,\text{Float}} = \frac{f_i Q_i P_i}{\sum_{j=1}^{N} f_j Q_j P_j}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:fundamental_weight}
\vspace{-0.4em}
\[
w_i^F = \frac{F_i}{\sum_{j=1}^{N} F_j}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:price_weight}
\vspace{-0.4em}
\[
w_i^P = \frac{P_i}{\sum_{j=1}^{N} P_j}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:price_weighted_value}
\vspace{-0.4em}
\[
V_{\text{PRI},t} = \frac{\sum_{i=1}^{N} P_{i,t}}{D_t}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:equal_weight}
\vspace{-0.4em}
\[
w_i^E = \frac{1}{N}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:equal_weighted_value}
\vspace{-0.4em}
\[
V_{\text{EQI},t} = \frac{\sum_{i=1}^{N} n_i P_{i,t}}{D_t}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Divisor Adjustment for Stock Splits}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:divisor_adjustment}
\vspace{-0.4em}

D_{\text{new}} = \frac{\sum P_{\text{new}}}{V_{\text{PRI, before}}}
\]
\end{tcolorbox}

\]
\end{tcolorbox}

\subsection*{Two-Stage Dividend Growth Model}

E[S_n] = \frac{Div_{n+1}}{r - g_l}
\]
\end{tcolorbox}

\subsection*{Implied Forward Interest Rates Using Cash Flow Additivity}

\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:total_prob_rule_expectation_m6}
\vspace{-0.4em}
\[
E[X] = \sum_{i=1}^n E[X | S_i] \times P(S_i)
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:law_of_total_variance_example}
\vspace{-0.4em}
\[
\text{Var}(R) = E[\text{Var}(R|S)] + \text{Var}(E[R|S])
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Conditional Means, Variances, and Covariances}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:conditional_covariance_formula_m6}
\vspace{-0.4em}
\[
\text{Cov}(X, Y | S) = E[(X - E[X|S])(Y - E[Y|S]) | S]
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Bayesian Updating}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:bayes_formula_m6}
\vspace{-0.4em}
\[
P(A|B) = \frac{P(B|A) \times P(A)}{P(B)}
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item $P(A|B)$ is the posterior probability of $A$ occurring given $B$ is true.
  \item $P(B|A)$ is the likelihood of observing $B$ given $A$ is true.
  \item $P(A)$ is the prior probability of $A$.
  \item $P(B)$ is the marginal probability of observing $B$, computed using the total probability rule:
\end{itemize}
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:total_prob_denominator_m6}
\vspace{-0.4em}
\[
P(B) = \sum_i P(B|A_i) \times P(A_i)
\]
\end{tcolorbox}

Z = \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \xrightarrow{d} N(0, 1) \quad \text{as } n \to \infty

\subsection*{Returns of a Portfolio}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_return_single_m8}
\vspace{-0.4em}
\[
R_{P,t} = \sum_{i=1}^N w_{i,t-1} R_{i,t}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:asset_realized_return_m8}
\vspace{-0.4em}
\[
R_{i,t} = \frac{P_{i,t} - P_{i,t-1} + \text{Div}_{i,t}}{P_{i,t-1}}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_cumulative_no_drift_m8}
\vspace{-0.4em}
\[
R_{P,T} = \sum_{i=1}^N w_{i,0} R_{i,T}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_arithmetic_average_m8}
\vspace{-0.4em}
\[
\bar{R}_P = \frac{1}{T} \sum_{t=1}^T R_{P,t}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_cumulative_drift_m8}
\vspace{-0.4em}
\[
R_{P,T} = \prod_{t=1}^T \left(1 + \sum_{i=1}^N w_{i,t-1} R_{i,t}\right) - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_geometric_return_m8}
\vspace{-0.4em}
\[
\bar{R}_{g,P} = \left[ \prod_{t=1}^T (1 + R_{P,t}) \right]^{\frac{1}{T}} - 1 = (1 + R_{P,T})^{\frac{1}{T}} - 1
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Variance and Standard Deviation of Portfolio Returns}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_variance_sum_m8}
\vspace{-0.4em}
\[
\sigma_P^2 = \sum_{i=1}^N \sum_{j=1}^N w_i w_j \sigma_{i,j}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:portfolio_expected_variance_m8}
\vspace{-0.4em}
\[
E[\sigma_P^2] = \sum_{i=1}^N \sum_{j=1}^N w_i w_j E[\sigma_{i,j}]
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:two_asset_variance_m8}
\vspace{-0.4em}
\[
\sigma_P^2 = w_1^2 \sigma_1^2 + w_2^2 \sigma_2^2 + 2 w_1 w_2 \sigma_{1,2} = w_1^2 \sigma_1^2 + w_2^2 \sigma_2^2 + 2 w_1 w_2 \rho_{1,2} \sigma_1 \sigma_2
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Matrix Notation of Portfolio Risk}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:unique_covariances_m8}
\vspace{-0.4em}
\[
\text{Unique Covariances} = \frac{N(N-1)}{2}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:covariance_matrix_m8}
\vspace{-0.4em}
\[
\mathbf{V} = \begin{bmatrix}
		\sigma_1^2 & \sigma_{1,2} & \sigma_{1,3} & \cdots & \sigma_{1,N} \\
		\sigma_{2,1} & \sigma_2^2 & \sigma_{2,3} & \cdots & \sigma_{2,N} \\
		\sigma_{3,1} & \sigma_{3,2} & \sigma_3^2 & \cdots & \sigma_{3,N} \\
		\vdots & \vdots & \vdots & \ddots & \vdots \\
		\sigma_{N,1} & \sigma_{N,2} & \sigma_{N,3} & \cdots & \sigma_N^2
	\end{bmatrix}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:correlation_matrix_m8}
\vspace{-0.4em}
\[
\boldsymbol{\rho} = \begin{bmatrix}
		1 & \rho_{1,2} & \rho_{1,3} & \cdots & \rho_{1,N} \\
		\rho_{2,1} & 1 & \rho_{2,3} & \cdots & \rho_{2,N} \\
		\rho_{3,1} & \rho_{3,2} & 1 & \cdots & \rho_{3,N} \\
		\vdots & \vdots & \vdots & \ddots & \vdots \\
		\rho_{N,1} & \rho_{N,2} & \rho_{N,3} & \cdots & 1
	\end{bmatrix}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Expected Return and Variance under Economic Scenarios}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:scenario_portfolio_return_m8}
\vspace{-0.4em}
\[
R_{P,M} = \sum_{i=1}^N w_i R_{i,M}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:expected_return_scenarios_m8}
\vspace{-0.4em}
\[
E[R_P] = \sum_{M=1}^M p_M R_{P,M} = \sum_{M=1}^M p_M \left( \sum_{i=1}^N w_i R_{i,M} \right)
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:variance_scenarios_m8}
\vspace{-0.4em}
\[
\sigma_P^2 = \sum_{M=1}^M p_M \left( R_{P,M} - E[R_P] \right)^2
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{The Benefits of Diversification}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:equal_weighted_variance_m8}
\vspace{-0.4em}
\[
\sigma_P^2 = \frac{\sigma^2}{N} + \left(1 - \frac{1}{N}\right) \bar{\rho} \sigma^2
\]
\noindent\textbf{Diễn giải:}
\begin{itemize}[leftmargin=1.5em, noitemsep, topsep=2pt, parsep=0pt]
  \item The contribution of the individual asset variance, $\frac{\sigma^2}{N}$, converges to $0$.
  \item The multiplier $\left(1 - \frac{1}{N}\right)$ approaches $1$.
\end{itemize}
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:limit_variance_m8}
\vspace{-0.4em}
\[
\lim_{N \to \infty} \sigma_P^2 = \bar{\rho} \sigma^2 = \text{Average Covariance}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{The Feasible Set and Efficient Frontier}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:optimization_sum_m8}
\vspace{-0.4em}
\[
\text{Minimize } \sigma_P^2 = \sum_{i=1}^N \sum_{j=1}^N w_i w_j \sigma_{i,j} \quad \text{subject to } \sum_{i=1}^N w_i = 1, \quad w_i \ge 0, \quad E[R_P] = k
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:optimization_matrix_m8}
\vspace{-0.4em}
\[
\text{Minimize } \sigma_P^2 = \mathbf{w}^T \mathbf{V} \mathbf{w} \quad \text{subject to } \mathbf{w}^T \mathbf{1} = 1, \quad \mathbf{w}^T \boldsymbol{\mu} = k, \quad \mathbf{w} \ge \mathbf{0}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Derivation of the Minimum-Variance Portfolio Weights}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:MVP_weight_A_m8}
\vspace{-0.4em}
\[
w_A^* = \frac{\sigma_B^2 - \sigma_{A,B}}{\sigma_A^2 + \sigma_B^2 - 2\sigma_{A,B}}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:MVP_weight_B_m8}
\vspace{-0.4em}
\[
w_B^* = 1 - w_A^* = \frac{\sigma_A^2 - \sigma_{A,B}}{\sigma_A^2 + \sigma_B^2 - 2\sigma_{A,B}}
\]
\end{tcolorbox}

\subsection*{Investor Utility and Indifference Curves}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:utility_function_m8}
\vspace{-0.4em}
\[
U = E[R] - \frac{1}{2} A \sigma^2
\]
\end{tcolorbox}

\subsection*{The Capital Allocation Line (CAL) and Sharpe Ratio}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:CAL_expected_return_m8}
\vspace{-0.4em}
\[
E[R_c] = w_f R_f + (1 - w_f) E[R_P]
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:CAL_std_m8}
\vspace{-0.4em}
\[
\sigma_c = (1 - w_f) \sigma_P \implies (1 - w_f) = \frac{\sigma_c}{\sigma_P}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:CAL_equation_m8}
\vspace{-0.4em}
\[
E[R_c] = R_f + \left( \frac{E[R_P] - R_f}{\sigma_P} \right) \sigma_c
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:sharpe_ratio_m8}
\vspace{-0.4em}
\[
\text{Sharpe Ratio} = \frac{E[R_P] - R_f}{\sigma_P}
\]
\end{tcolorbox}

\subsection*{The Capital Market Line (CML)}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:CML_equation_m8}
\vspace{-0.4em}
\[
E[R_P] = R_f + \left( \frac{E[R_M] - R_f}{\sigma_M} \right) \sigma_P
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{The Capital Asset Pricing Model (CAPM) and Beta}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:beta_definition_m8}
\vspace{-0.4em}
\[
\beta_i = \frac{\sigma_{i,M}}{\sigma_M^2} = \frac{\rho_{i,M} \sigma_i \sigma_M}{\sigma_M^2} = \rho_{i,M} \frac{\sigma_i}{\sigma_M}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:CAPM_equation_m8}
\vspace{-0.4em}
\[
E[R_i] = R_f + \beta_i \left( E[R_M] - R_f \right)
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:SML_covariance_form_m8}
\vspace{-0.4em}
\[
E[R_i] = R_f + \left(\frac{E[R_M] - R_f}{\sigma_M^2}\right) \sigma_{i,M}
\]
\end{tcolorbox}

\subsection*{Estimating Beta using Linear Regression}

\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]

\section*{Module 10: Applications of Simple Linear Regression in Finance}

\subsection*{Linear Regression Analysis}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:pop_regression_m10}
\vspace{-0.4em}
\[
Y_i = \beta_0 + \beta_1 X_i + \epsilon_i, \quad i = 1, \dots, n
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:est_regression_m10}
\vspace{-0.4em}
\[
\hat{Y}_i = \hat{\beta}_0 + \hat{\beta}_1 X_i
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Coefficient Estimation of Linear Regression}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:sse_definition_m10}
\vspace{-0.4em}
\[
\text{SSE} = \sum_{i=1}^n (Y_i - \hat{Y}_i)^2 = \sum_{i=1}^n e_i^2
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:slope_estimator_m10}
\vspace{-0.4em}
\[
\hat{\beta}_1 = \frac{\text{Cov}(X, Y)}{\text{Var}(X)} = \frac{\sum_{i=1}^n (X_i - \bar{X})(Y_i - \bar{Y})}{\sum_{i=1}^n (X_i - \bar{X})^2}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:intercept_estimator_m10}
\vspace{-0.4em}
\[
\hat{\beta}_0 = \bar{Y} - \hat{\beta}_1 \bar{X}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Estimation Residuals}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:residual_definition_m10}
\vspace{-0.4em}
\[
e_i = Y_i - \hat{Y}_i
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Evaluating Goodness-of-Fit and Regression Coefficients}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:sst_decomposition_m10}
\vspace{-0.4em}
\[
\text{SST} = \text{SSR} + \text{SSE}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:r2_definition_m10}
\vspace{-0.4em}
\[
R^2 = \frac{\text{SSR}}{\text{SST}} = 1 - \frac{\text{SSE}}{\text{SST}}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:correlation_r2_m10}
\vspace{-0.4em}
\[
r = \text{sign}(\hat{\beta}_1) \times \sqrt{R^2}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Hypothesis Testing and ANOVA Estimates}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:see_definition_m10}
\vspace{-0.4em}
\[
s_e = \sqrt{\text{MSE}} = \sqrt{\frac{\text{SSE}}{n - 2}}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:t_stat_slope_m10}
\vspace{-0.4em}
\[
t = \frac{\hat{\beta}_1 - 0}{s_{\hat{\beta}_1}}
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:se_slope_m10}
\vspace{-0.4em}
\[
s_{\hat{\beta}_1} = \frac{s_e}{\sqrt{\sum_{i=1}^n (X_i - \bar{X})^2}}
\]
\end{tcolorbox}
\vspace{0.3em}
\subsection*{Prediction Using a Simple Regression}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:predicted_value_m10}
\vspace{-0.4em}
\[
\hat{Y}_p = \hat{\beta}_0 + \hat{\beta}_1 X_p
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:variance_prediction_error_m10}
\vspace{-0.4em}
\[
s_{pred}^2 = s_e^2 \left[ 1 + \frac{1}{n} + \frac{(X_p - \bar{X})^2}{\sum_{i=1}^n (X_i - \bar{X})^2} \right]
\]
\end{tcolorbox}
\vspace{0.3em}
\begin{tcolorbox}[colback=blue!3!white, colframe=trustvilleblue!80!black, boxrule=0.5pt, arc=3pt, left=6pt, right=6pt, top=4pt, bottom=4pt, breakable]
\noindent\textbf{Công thức} \eqref{eq:prediction_interval_m10}
\vspace{-0.4em}
\[
\hat{Y}_p \pm t_{crit} \times s_{pred}
\]
\end{tcolorbox}
